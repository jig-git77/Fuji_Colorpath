			,@CMYK_K

            ,@LAB_b_ACT

CREATE PROCEDURE [dbo].[uspInsertCPColorMeasurement]
 @CPMeasurementGUID			uniqueidentifier
,@Originator				varchar(50)
,@MeasurementDate			datetime	
,@MeasurementGMT			datetime
,@Target_Name				nvarchar(50)	= NULL
,@Proof_Supplier			nvarchar(50)	= NULL
,@Target_File				nvarchar(250)	= NULL 
,@CPSerialNUM				nvarchar(50)	= NULL 
,@CPVClientMID				int				= NULL

,@deltaEType				tinyint

--*************Target Tolerance
,@Targ_Paper_dE				decimal(6,2)
,@Targ_Avg_dE				decimal(6,2)
,@Targ_Maximum_dE			decimal(6,2)	= NULL
,@Targ_Primary_dE			decimal(6,2)	= NULL
,@Targ_Primary_DH			decimal(6,2)	= NULL
,@Targ_Gray_DH				decimal(6,2)	= NULL
,@Targ_95PCT_Quantile		decimal(6,2)	= NULL
,@Targ_Avg_OuterArea		decimal(6,2)	= NULL

--************ 2 Level Target Tolerance
,@TargStrict_Paper_dE		decimal(6,2)	= NULL
,@TargStrict_Avg_dE			decimal(6,2)	= NULL
,@TargStrict_Maximum_dE		decimal(6,2)	= NULL
,@TargStrict_Primary_dE     decimal(6,2)	= NULL
,@TargStrict_Primary_DH		decimal(6,2)	= NULL
,@TargStrict_Gray_DH		decimal(6,2)	= NULL


--Measured Values
,@Paper_dE					decimal(6,2)	
,@AVG_dE					decimal(6,2)
,@MAX_dE					decimal(6,2)	= NULL
,@PatchMaxdE				varchar(12)		= NULL
,@Primary_dE				decimal(6,2)	= NULL
,@PatchPrimarydE			varchar(12)     = NULL
,@MAX_Primary_DH			decimal(6,2)	= NULL
,@PatchPrimaryDH			varchar(12)     = NULL
,@AVG_Gray_DH				decimal(6,2)	= NULL
,@95PCT_Quantile		    decimal(6,2)	= NULL
,@Avg_OuterArea			    decimal(6,2)	= NULL 


,@User						nvarchar(50)	= NULL
,@Customer					nvarchar(50)	= NULL
,@Job						nvarchar(50)	= NULL
,@Info_Text					nvarchar(250)	= NULL
,@Proofer					nvarchar(50)	= NULL		        
,@Status					tinyint			= NULL

,@PrinterMake				nvarchar(50)    = NULL
,@PrinterModel				nvarchar(50)    = NULL
,@PrinterSerialNum			nvarchar(50)    = NULL	
,@RipType					nvarchar(50)    = NULL

--Measuring Device
,@Instrument				nvarchar(50)	= NULL
,@Instrument_SerialNum		nvarchar(20)	= NULL
,@Illuminant				nvarchar(20)	= NULL
,@Filter					nvarchar(20)	= NULL 
,@ObserverType				nvarchar(20)	= NULL 

,@Substrate					nvarchar(50)	= NULL
,@Substrate_SKU				nvarchar(50)	= NULL
,@Substrate_Batch			nvarchar(50)	= NULL			
,@Ink						nvarchar(50)	= NULL
,@Ink_SKU					nvarchar(50)	= NULL
,@Ink_Batch					nvarchar(50)	= NULL
,@Chemistry					nvarchar(50)	= NULL
,@Chemistry_SKU				nvarchar(50)	= NULL
,@Chemistry_Batch			nvarchar(50)	= NULL

,@OutputCategory		    varchar(50)     = NULL
,@OutputSeparation			varchar(30)		= NULL
,@OutputResolution			decimal(6,2)	= NULL
,@OutputLineScreen			decimal(6,2)	= NULL
,@OutputDotShape			varchar(30)		= NULL

--Begin Print Target Properties
,@TargDensity_C				decimal(6,2)	= NULL
,@TargDensity_M				decimal(6,2)	= NULL
,@TargDensity_Y				decimal(6,2)	= NULL
,@TargDensity_K				decimal(6,2)	= NULL
,@TargDensity_Th			decimal(6,2)	= NULL
,@TargDotGain_C				decimal(6,2)	= NULL
,@TargDotGain_M				decimal(6,2)	= NULL
,@TargDotGain_Y				decimal(6,2)	= NULL
,@TargDotGain_K				decimal(6,2)	= NULL
,@TargDotGain_Th			decimal(6,2)	= NULL
,@TargPrintContrast_C		decimal(6,2)	= NULL
,@TargPrintContrast_M		decimal(6,2)	= NULL
,@TargPrintContrast_Y		decimal(6,2)	= NULL
,@TargPrintContrast_K		decimal(6,2)	= NULL
,@TargPrintContrast_Th		decimal(6,2)	= NULL
,@TargTrap_R				decimal(6,2)	= NULL
,@TargTrap_G				decimal(6,2)	= NULL
,@TargTrap_B				decimal(6,2)	= NULL
,@TargTrap_Th				decimal(6,2)	= NULL


,@NumOfPatches				smallint			
,@NumOfPatchLines			smallint
AS
-- ===================================================
-- Author: Chirag Shah
-- Create date: 11th November - 2009
-- Description:	Insert Plate Measurement  
-- ===================================================
BEGIN

	SET NOCOUNT ON
    
	DECLARE @ErrorMSg varchar(500)
	DECLARE @CPDongleID		 int
	DECLARE @CPDeviceID		 int

	DECLARE @LastMeasurementID int
	SET     @LastMeasurementID  = 0

	DECLARE @CPDevicePrintTargetID int  --Print Target ID
	/****** Begin Plate Measurement ************/
	DECLARE @DeviceTypeID	 tinyint 
	SET 	@DeviceTypeID = 1

	DECLARE @MeasurementTypeID	tinyint
	SET @MeasurementTypeID = 1
	/****** End Plate Measurement  ************/

	BEGIN TRY
		
		BEGIN TRAN		

		-- check whether it is valid and active dongle 
		IF NOT EXISTS (SELECT * FROM CPDongle WHERE CPSerialNUM = @CPSerialNUM AND IsUsed = 1 ) 
		BEGIN
			SET @ErrorMSg = 'Dongle ' + ISNULL(@CPSerialNUM,'') + ' Does Not Exists in CPDongle Table'	
			RAISERROR(@ErrorMSg,16,1)
		END

		--Find Dongle ID
		SELECT @CPDongleID = CPDongleID
		FROM CPDongle 
		WHERE CPSerialNUM = @CPSerialNUM
		
		-- check whether device already exist if not create a new device entry into CPDevice table 
		IF NOT EXISTS (  SELECT * FROM CPDevice WHERE  CPDeviceDesc = LTRIM(RTRIM((@Proofer)))  AND  CPDongleID = @CPDongleID AND MeasurementTypeID = 1 )
		BEGIN
			INSERT INTO CPDevice (CPDongleID,CPDeviceDesc,MeasurementTypeID, CPDeviceTypeID, LastMeasurementID) VALUES (@CPDongleID, LTRIM(RTRIM((@Proofer))),@MeasurementTypeID,@DeviceTypeID,0 )
			SET @CPDeviceID = SCOPE_IDENTITY()
		END
		Else
			SELECT @CPDeviceID = CPDeviceID,@LastMeasurementID=LastMeasurementID FROM CPDevice WHERE CPDeviceDesc = LTRIM(RTRIM(@Proofer)) AND CPDongleID = @CPDongleID AND MeasurementTypeID = @MeasurementTypeID
	
		SET @LastMeasurementID = @LastMeasurementID  + 1

		UPDATE CPDevice 
		SET LastMeasurementID = @LastMeasurementID 
		WHERE CPDeviceID = @CPDeviceID

		--Checke whether it is a new print target for that device, if it is new insert an entry into CPDevicePrintTarget table
		IF NOT EXISTS (SELECT * FROM CPDevicePrintTarget WHERE PrintTarget = @Target_Name AND DeviceID = @CPDeviceID ) 
		BEGIN
			INSERT INTO [CPDevicePrintTarget] (DeviceID,PrintTarget) VALUES (@CPDeviceID,@Target_Name)
			SET @CPDevicePrintTargetID = SCOPE_IDENTITY()
		END
		ELSE --else retrieve PrintTargetID for that Device
		BEGIN
			SELECT @CPDevicePrintTargetID=CPDevicePrintTargetID FROM CPDevicePrintTarget 
			WHERE PrintTarget=@Target_Name AND DeviceID = @CPDeviceID
		END

		
		-- Insert data into CPMeasurement 
		INSERT INTO [CPMeasurement]
		(
			 [CPMeasurementGUID]
			,[CPMeasurementID]
			,[CPVClientMID]
			,[CPDongleID]
			,[CPDeviceID]
			,[Originator]
			,[CreatedDT]
			,[CreatedGMT]
			,[User]
			,[Customer]
			,[Job]
			,[Info_Text]
			,[Status]
			,[NumOfPatches]
			,[NumOfPatchLines]
			,[CPMeasurementTypeID]
		)
		VALUES
		(
			 @CPMeasurementGUID
			,@LastMeasurementID
			,@CPVclientMID
			,@CPDongleID
			,@CPDeviceID
			,@Originator	
			,@MeasurementDate
			,@MeasurementGMT
			,@User
			,@Customer
			,@Job
			,@Info_Text
			,@Status
			,@NumOfPatches
			,@NumOfPatchLines
			,@MeasurementTypeID
		)	

		--Dot Area Tolerances
		INSERT INTO [dbo].[CPMeasurement_TOL]
           ([CPMeasurementGUID]
           ,[Paper_Limit]
           ,[Average_Limit]
           ,[Maximum_Limit]
           ,[Primary_Limit]
           ,[PRIMARY_LIMIT_DH]
           ,[GRAY_AVERAGE_LIMIT_DH]
		   ,[Tol_Limit_95pct_Patches]
           ,[Tol_Limit_OuterGamut]	
           ,[Paper_Strict]
           ,[Average_Strict]
           ,[Maximum_Strict]
           ,[Primary_Strict]
           ,[Primary_Max_DH]
           ,[Gray_Avg_Max_DH]
		   ,[deltaEForumlaID]    /* missing delta e type */	
          )
     VALUES
           (
			 @CPMeasurementGUID	
			,@Targ_Paper_dE
			,@Targ_Avg_dE
            ,@Targ_Maximum_dE
			,@Targ_Primary_dE
            ,@Targ_Primary_DH
			,@Targ_Gray_DH
            ,@Targ_95PCT_Quantile
            ,@Targ_Avg_OuterArea
            ,@TargStrict_Paper_dE
            ,@TargStrict_Avg_dE
            ,@TargStrict_Maximum_dE
            ,@TargStrict_Primary_dE
			,@TargStrict_Primary_DH	
			,@TargStrict_Gray_DH
			,@deltaEType 		
          
			)

		--Measured Values
		INSERT INTO [dbo].[CPMeasurement_ACT]
           (
			[CPMeasurementGUID]
           ,[Paper]
           ,[Average]
           ,[PatchMaximumdE]
           ,[PatchdEValue]
           ,[PatchPrimarydE]
           ,[PatchPrimarydEValue]
           ,[Primary_Max_DH]
           ,[Primary_Max_DH_Patch]
           ,[Gray_Average_DH]
           ,[dEAvg_95pct_Patches]
           ,[dEAvg_OuterGamut]
			)
		 VALUES
           (
			 @CPMeasurementGUID	
            ,@Paper_dE
            ,@AVG_dE
            ,@PatchMaxdE
			,@MAX_dE
            ,@PatchPrimarydE
            ,@Primary_dE
            ,@MAX_Primary_DH
            ,@PatchPrimaryDH
            ,@AVG_Gray_DH
            ,@95PCT_Quantile
            ,@Avg_OuterArea
		   	
		  )

		--Insert into Print Target
		INSERT INTO [CPMeasurement_PT]
		(
			[CPMeasurementGUID]
		   ,[CPDevicePrintTargetID]
           ,[Proof_Supplier]
		   ,[Target_File]
		)
		VALUES
		(
			@CPMeasurementGUID
		   ,@CPDevicePrintTargetID
           ,@Proof_Supplier
           ,@Target_File 
		)	

		--Measuring Instrument
		INSERT INTO [CPMeasurement_MD]
		(
			 [CPMeasurementGUID]
			,[Instrument]
			,[DeviceSerialNum]
			,[ObserverType]
			,[Illuminant]
			,[Filter]
		)
		VALUES
		(
			@CPMeasurementGUID
		   ,@Instrument	
		   ,@Instrument_SerialNum
           ,@ObserverType
           ,@Illuminant
           ,@Filter	
		)

		--Insert into Output Device
		INSERT INTO [CPMeasurement_OD]
		(
			[CPMeasurementGUID]
			,[Category]
			,[Separation]  --For Plates Only
			,[Manufacturer]
			,[Model]
			,[Serial_Num]
			,[RIP_Type]
		)
		VALUES
		(
			 @CPMeasurementGUID 
			,@OutputCategory
			,@OutputSeparation
			,@PrinterMake 
			,@PrinterModel  
			,@PrinterSerialNum
			,@RIPType
		)		

		--Insert Device Parameters
		INSERT INTO [CPMeasurement_DP]
		(
			[CPMeasurementGUID],
			[Substrate],
			[Substrate_SKU],
			[Substrate_Batch],
			[Ink],
			[Ink_SKU],
			[Ink_Batch],
			[Chemistry],
			[Chemistry_SKU],
			[Chemistry_Batch],
			[Resolution],
			[Line_Screen],
			[Dot_Shape]
		)
		VALUES
		(
			@CPMeasurementGUID,
			@Substrate,
			@Substrate_SKU,
			@Substrate_Batch,
			@Ink,
			@Ink_SKU,
			@Ink_Batch,
			@Chemistry,
			@Chemistry_SKU,
			@Chemistry_Batch,
			@OutputResolution,
			@OutputLineScreen,
			@OutputDotShape
		)	


		DECLARE @TempRHDeviceID	smallint

		SELECT TOP 1 @TempRHDeviceID=TempRHDeviceID	
		FROM  TempRHDevice_CPDevice WHERE CPDeviceID = @CPDeviceID

		IF ((SELECT COUNT(*) 
			 FROM TempRHData 
			 INNER JOIN TempRHDevice ON TempRHData.TempRHDeviceID = TempRHDevice.TempRHDeviceID
			 WHERE TempRHData.TempRHDeviceID = @TempRHDeviceID
	         AND CreatedDT >= DATEADD(mi,-30,@MeasurementDate)
	         AND CreatedDT <  DATEADD(mi, 30,@MeasurementDate)) > 0 )
		     BEGIN 
				INSERT INTO [dbo].[CPMeasurement_TempRH]
				   ([CPMeasurementGUID]
				   ,[TempC]
				   ,[TempF]
				   ,[RH])
				SELECT TOP 1
				@CPMeasurementGUID,
				TempC,
				TempF,
				RH
				FROM TempRHData 
				INNER JOIN TempRHDevice ON TempRHData.TempRHDeviceID = TempRHDevice.TempRHDeviceID
				WHERE TempRHData.TempRHDeviceID = @TempRHDeviceID
				AND CreatedDT >= DATEADD(mi,-15,@MeasurementDate)
				AND CreatedDT <  DATEADD(mi, 15,@MeasurementDate)
				ORDER BY ABS(DATEDIFF(SS, CreatedDT,@MeasurementDate))
			END

		COMMIT TRAN


	END TRY
	BEGIN CATCH
		-- Whoops, there was an error
		-- Rollback any active or uncommittable transactions before
		-- inserting information in the ErrorLog
		IF @@TRANCOUNT > 0
			ROLLBACK
			EXECUTE [dbo].[uspLogError]; 		
			
			DECLARE @ErrorMessage NVARCHAR(4000);
			DECLARE @ErrorSeverity INT;
			DECLARE @ErrorState INT;
			
			SELECT 
				@ErrorMessage = '- INSERT ERROR CPV Color Measurement ' + ERROR_MESSAGE(),
				@ErrorSeverity = ERROR_SEVERITY(),
				@ErrorState = ERROR_STATE();
			RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState)

	END CATCH

END
